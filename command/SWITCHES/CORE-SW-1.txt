! =========================================
! 10. ACLs (explicit sequence numbers)
! =========================================
! Defined BEFORE the interfaces that use them.
! IT (192.168.10.0/24) is not blocked by DEPT-ISOLATION; TCP established
! and ICMP echo-reply to IT are permitted before department denies.
! In SERVER-ACCESS, line 56 permits ICMP echo (ping requests) toward
! servers (172.16.10.0/24) before line 130 deny.

ip access-list extended DEPT-ISOLATION
 10 permit tcp any 192.168.10.0 0.0.0.255 established
 20 permit icmp any 192.168.10.0 0.0.0.255 echo-reply
 30 deny ip any 192.168.10.0 0.0.0.255
 40 deny ip any 192.168.20.0 0.0.0.255
 50 deny ip any 192.168.30.0 0.0.0.255
 60 deny ip any 192.168.40.0 0.0.0.255
 70 deny ip any 192.168.50.0 0.0.0.255
 80 deny ip any 192.168.60.0 0.0.0.255
 90 deny ip any 192.168.70.0 0.0.0.255
 100 deny ip any 192.168.80.0 0.0.0.255
 110 deny ip any 192.168.90.0 0.0.0.255
 120 deny ip any 192.168.100.0 0.0.0.255
 130 deny ip any 192.168.110.0 0.0.0.255
 140 deny ip any 192.168.120.0 0.0.0.255
 150 deny ip any 192.168.130.0 0.0.0.255
 160 deny ip any 192.168.140.0 0.0.0.255
 170 permit ip any any
exit

ip access-list extended MGMT-ONLY
 10 permit ip 192.168.10.0 0.0.0.255 192.168.99.0 0.0.0.255
 20 deny ip any 192.168.99.0 0.0.0.255

exit

ip access-list extended SERVER-ACCESS

 40 permit udp 192.168.0.0 0.0.255.255 host 172.16.10.10 eq domain
 50 permit tcp 192.168.0.0 0.0.255.255 host 172.16.10.10 eq domain
 55 permit tcp 192.168.0.0 0.0.255.255 172.16.10.0 0.0.0.255 established
 56 permit icmp 192.168.0.0 0.0.255.255 172.16.10.0 0.0.0.255 echo
 57 permit tcp host 192.168.130.10 host 172.16.10.13 eq www
 58 permit tcp host 192.168.130.10 host 172.16.10.13 eq 443
59 permit icmp 192.168.0.0 0.0.255.255 172.16.10.0 0.0.0.255 echo-reply
 60 deny ip 192.168.130.0 0.0.0.255 172.16.10.0 0.0.0.255
 70 permit tcp 192.168.0.0 0.0.255.255 host 172.16.10.12 eq ftp
 80 permit tcp 192.168.0.0 0.0.255.255 host 172.16.10.12 eq 20
 90 permit tcp 192.168.0.0 0.0.255.255 host 172.16.10.12 eq 445
 100 permit tcp 192.168.0.0 0.0.255.255 host 172.16.10.12 eq 139
 110 permit tcp 192.168.0.0 0.0.255.255 host 172.16.10.13 eq www
