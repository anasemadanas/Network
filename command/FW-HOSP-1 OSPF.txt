! =========================================
! 3. ACLs & VPN INTERFACE BINDING
! =========================================
access-list VPN-CLINIC extended permit ip 172.16.10.0 255.255.255.0 192.168.2.0 255.255.255.0
access-list VPN-CLINIC extended permit ip host 172.16.10.12 192.168.2.0 255.255.255.0

access-group HOSP-VPN-IN in interface ORANGE-VPN

! =========================================
! 4. GROUP POLICY & IPSEC VPN
! =========================================
group-policy DOCTORS internal

crypto map HOSP-MAP 10 match address VPN-CLINIC
crypto map HOSP-MAP 10 set peer 100.1.3.2
crypto map HOSP-MAP interface ORANGE-VPN

crypto ikev1 enable ORANGE-VPN

crypto ikev1 policy 10
 encr aes
 authentication pre-share
 group 2
exit

tunnel-group 100.1.3.2 type ipsec-l2l
tunnel-group 100.1.3.2 ipsec-attributes
 ikev1 pre-shared-key ClinicHospVPN2026!
exit

  inspect ftp

  inspect tftp
